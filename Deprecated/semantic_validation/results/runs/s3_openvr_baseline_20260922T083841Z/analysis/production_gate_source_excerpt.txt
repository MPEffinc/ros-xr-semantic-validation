    45	        for i in range(openvr.k_unMaxTrackedDeviceCount):
    46	            if self.vr.getTrackedDeviceClass(i) == openvr.TrackedDeviceClass_Controller:
    47	                role = self.vr.getControllerRoleForTrackedDeviceIndex(i)
    48	                
    49	                if role == openvr.TrackedControllerRole_RightHand and poses[i].bPoseIsValid:
    50	                    
    51	                    # 1. Grip Button Check
    52	                    result, state = self.vr.getControllerState(i)
    53	                    is_gripping = bool(state.ulButtonPressed & (1 << openvr.k_EButton_Grip))
    54	
    55	                    if not is_gripping:
    56	                        if not self.first_packet:
    57	                            self.get_logger().info("Teleop Disengaged.")
    58	                        self.first_packet = True 
    59	                        return
    60	
    61	                    # 2. Extract Data
    62	                    matrix = poses[i].mDeviceToAbsoluteTracking
    63	                    raw_x, raw_y, raw_z = float(matrix[0][3]), float(matrix[1][3]), float(matrix[2][3])
    64	                    
    65	                    current_rot_mtx = [
    66	                        [matrix[0][0], matrix[0][1], matrix[0][2]],
    67	                        [matrix[1][0], matrix[1][1], matrix[1][2]],
    68	                        [matrix[2][0], matrix[2][1], matrix[2][2]]
    69	                    ]
    70	                    current_rot = R.from_matrix(current_rot_mtx)
    71	
    72	                    # 3. Calibration on Grip
    73	                    if self.first_packet:
    74	                        self.offset_x, self.offset_y, self.offset_z = raw_x, raw_y, raw_z
    75	                        self.offset_rot_inv = current_rot.inv()
    76	                        self.first_packet = False
    77	                        self.get_logger().info("Teleop Engaged: Relative tracking active.")
    78	
    79	                    # 4. Construct Message
    80	                    msg = PoseStamped()
    81	                    msg.header.stamp = self.get_clock().now().to_msg()
    82	                    msg.header.frame_id = "base_link" 
    83	
    84	                    # POSITION LOGIC
    85	                    # Moves relative to a 40cm forward / 30cm up starting point
    86	                    msg.pose.position.x = ((raw_x - self.offset_x) * self.scaling) + 0.4
    87	                    msg.pose.position.y = ((raw_y - self.offset_y) * self.scaling)
    88	                    msg.pose.position.z = ((raw_z - self.offset_z) * self.scaling) + 0.3
    89	
    90	                    # ROTATION LOGIC (The Fix)
    91	                    # Hand Relative Change = Initial_Hand_Inv * Current_Hand
    92	                    hand_relative_rotation = self.offset_rot_inv * current_rot
    93	                    
    94	                    # Target = Robot_Home * Hand_Relative_Change
    95	                    final_rot = self.robot_home_rot * hand_relative_rotation
    96	                    quat = final_rot.as_quat()
    97	
    98	                    msg.pose.orientation.x = quat[0]
    99	                    msg.pose.orientation.y = quat[1]
   100	                    msg.pose.orientation.z = quat[2]
   101	                    msg.pose.orientation.w = quat[3]
   102	
   103	                    self.publisher_.publish(msg)
   104	
   105	def main(args=None):
   106	    rclpy.init(args=args)
   107	    node = QuestTeleop()
   108	    try:
   109	        rclpy.spin(node)
   110	    except KeyboardInterrupt:
   111	        pass
   112	    finally:
   113	        openvr.shutdown()
   114	        node.destroy_node()
   115	        rclpy.shutdown()
